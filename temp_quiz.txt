import { devLog, devError } from '@/utils/logger';
import { useUserStore } from '@/lib/stores/userStore';
import { setAvailablePoints as cachePoints } from '@/constants/points';

/**

function isQuizContent(data: unknown): data is QuizContent {
  if (!data || typeof data !== 'object') return false;

  const quiz = data as Record<string, unknown>;

  // ?„ìˆ˜ ?„ë“œ ê²€ì¦?
  if (typeof quiz.id !== 'string') return false;
  if (typeof quiz.title !== 'string') return false;
  if (typeof quiz.prompt !== 'string') return false;
  if (typeof quiz.correctAnswerId !== 'string') return false;

  // options ë°°ì—´ ê²€ì¦?
  if (!Array.isArray(quiz.options)) return false;

  // ê°??µì…˜??êµ¬ì¡° ê²€ì¦?
  return quiz.options.every((option: unknown) => {
    if (!option || typeof option !== 'object') return false;
    const opt = option as Record<string, unknown>;
    return typeof opt.id === 'string' && typeof opt.text === 'string';
  });
}

function QuizContent() {
  const { setTitle } = useScenarioHeader();
  const searchParams = useSearchParams();
  const quizId = searchParams.get('id') || '1'; // ê¸°ë³¸ê°?1

  const [quiz, setQuiz] = useState<QuizContent | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [hasDeposited, setHasDeposited] = useState(false);
  const [isDepositing, setIsDepositing] = useState(false);
  const [depositMessage, setDepositMessage] = useState<string | null>(null);
  const { availablePoints, setAvailablePoints } = useUserStore();

  useEffect(() => {
    setTitle('Quiz');
    return () => setTitle('');
  }, [setTitle]);

  useEffect(() => {
    setHasDeposited(false);
    setDepositMessage(null);
    setIsDepositing(false);
  }, [quizId]);

  useEffect(() => {
    async function loadQuiz() {
      try {
        setIsLoading(true);
        setError(null);

        // JSON ?Œì¼?ì„œ ?´ì¦ˆ ?°ì´??ë¡œë“œ
        const response = await fetch('/data/quizzes.json');

        if (!response.ok) {
          throw new Error(`?´ì¦ˆ ?°ì´??ë¡œë“œ ?¤íŒ¨: ${response.status}`);
        }

        const data = await response.json();

        // ?°ì´??êµ¬ì¡° ê²€ì¦?
        if (!data || typeof data !== 'object' || !Array.isArray(data.quizzes)) {
          throw new Error('?´ì¦ˆ ?°ì´???•ì‹???¬ë°”ë¥´ì? ?ŠìŠµ?ˆë‹¤.');
        }

        // IDë¡??´ì¦ˆ ì°¾ê¸° (?€???¨ì–¸ ?†ì´)
        const foundQuiz = data.quizzes.find((q: unknown) => {
          return q && typeof q === 'object' && 'id' in q && q.id === quizId;
        });

        if (!foundQuiz) {
          throw new Error(`?´ì¦ˆ ID ${quizId}ë¥?ì°¾ì„ ???†ìŠµ?ˆë‹¤.`);
        }

        // ?€??ê°€?œë¡œ ê²€ì¦?
        if (!isQuizContent(foundQuiz)) {
          devError('[QuizPage] ?´ì¦ˆ ?°ì´??êµ¬ì¡°ê°€ ?¬ë°”ë¥´ì? ?ŠìŒ:', foundQuiz);
          throw new Error('?´ì¦ˆ ?°ì´??êµ¬ì¡°ê°€ ?¬ë°”ë¥´ì? ?ŠìŠµ?ˆë‹¤.');
        }

        devLog(`[QuizPage] ?´ì¦ˆ ${quizId} ë¡œë“œ ?„ë£Œ`);
        setQuiz(foundQuiz);
      } catch (err) {
        devError('[QuizPage] ?´ì¦ˆ ë¡œë“œ ?¤íŒ¨:', err);
        setError(
          err instanceof Error ? err.message : '?´ì¦ˆë¥?ë¶ˆëŸ¬?????†ìŠµ?ˆë‹¤.'
        );
      } finally {
        setIsLoading(false);
      }
    }

    loadQuiz();
  }, [quizId]);

  const handleSelectOption = useCallback(
    async (selectedQuizId: string, optionId: string) => {
      if (!quiz || quiz.id !== selectedQuizId) return;
      if (quiz.id !== '2') return; // ?´ë‹¹ ?œë‚˜ë¦¬ì˜¤ ?´ì¦ˆ?ë§Œ ?¬ì¸???ë¦½
      if (hasDeposited || isDepositing) return;

      if (optionId !== quiz.correctAnswerId) {
        setDepositMessage('?•ë‹µ??? íƒ?˜ë©´ ?¬ì¸?¸ê? ?ë¦½?¼ìš”.');
        return;
      }

      try {
        setIsDepositing(true);
        setDepositMessage(null);

        const scenarioId = '2';
        const res = await axiosInstance.post(
          `/users/me/scenarios/${scenarioId}/reward`
        );
        const rewarded = res.data?.data?.rewarded;
        const rewardAmount = Number(
          res.data?.data?.amount ?? res.data?.data?.points ?? 1000
        );

        if (rewarded) {
          const newPoints = availablePoints + rewardAmount;
          setAvailablePoints(newPoints);
          cachePoints(newPoints);
        }

        setHasDeposited(true);
        setDepositMessage(
          rewarded
            ? '?œë‚˜ë¦¬ì˜¤ ?„ë£Œ! 1000?¬ì¸???ë¦½ ?„ë£Œ.'
            : '?´ë? ë³´ìƒ??ë°›ì•˜?µë‹ˆ??'
        );
        devLog('[QuizPage] ?¬ì¸???ë¦½ ?„ë£Œ ?¬ë?:', rewarded);
      } catch (err) {
        setHasDeposited(false);
        setDepositMessage('?¬ì¸???ë¦½???¤íŒ¨?ˆìŠµ?ˆë‹¤. ?¤ì‹œ ?œë„?´ì£¼?¸ìš”.');
        devError('[QuizPage] ?¬ì¸???ë¦½ ?¤íŒ¨:', err);
      } finally {
        setIsDepositing(false);
      }
    },
    [quiz, hasDeposited, isDepositing]
  );

  if (isLoading) {
    return (
      <div className="flex h-screen items-center justify-center">
        <p className="text-gray-500">?´ì¦ˆë¥?ë¶ˆëŸ¬?¤ëŠ” ì¤?..</p>
      </div>
    );
  }

  if (error || !quiz) {
    return (
      <div className="flex h-screen items-center justify-center">
        <p className="text-gray-500">{error || '?´ì¦ˆë¥?ë¶ˆëŸ¬?????†ìŠµ?ˆë‹¤.'}</p>
      </div>
    );
  }

  return (
    <div className="flex flex-col gap-[24px] px-[20px] py-[32px]">
      <QuizScreen quiz={quiz} onSelectOption={handleSelectOption} />
      {depositMessage && (
        <p className="text-center text-sm text-gray-700">{depositMessage}</p>
      )}
    </div>
  );
}

export default function QuizPage() {
  return (
    <Suspense
      fallback={
        <div className="flex h-screen items-center justify-center">
          <p className="text-gray-500">?´ì¦ˆë¥?ë¶ˆëŸ¬?¤ëŠ” ì¤?..</p>
        </div>
      }
    >
      <QuizContent />
    </Suspense>
  );
}
